  <script>
    // Find the preview, headings, controls, and copy interface.
    const heroPreview = document.getElementById("hero-preview");
    const heroHeading = document.getElementById("hero-heading");
    const heroSubheading = document.getElementById("hero-subheading");

    const darkAppearanceCheckbox =
      document.getElementById("dark-appearance");

    const centreTextCheckbox =
      document.getElementById("centre-text");

    const minimumHeightCheckbox =
      document.getElementById("minimum-height");

    const showSubheadingCheckbox =
      document.getElementById("show-subheading");

    const h1SerifCheckbox = document.getElementById("h1-serif");
    const h2SerifCheckbox = document.getElementById("h2-serif");

    const h1LetterSpacingSlider =
      document.getElementById("h1-letter-spacing");

    const h2LetterSpacingSlider =
      document.getElementById("h2-letter-spacing");

    const h1SpacingValue = document.getElementById("h1-spacing-value");
    const h2SpacingValue = document.getElementById("h2-spacing-value");

    const heroStylesElement = document.getElementById("hero-styles");
    const heroEffectStylesElement =
      document.getElementById("hero-effect-styles");

    const copyHtmlButton = document.getElementById("copy-html-button");
    const copyStatus = document.getElementById("copy-status");
    const fallbackPanel = document.getElementById("fallback-panel");
    const generatedHtmlTextarea = document.getElementById("generated-html");

    // Store all settings, including numeric slider values.
    const settings = {
      darkAppearance: darkAppearanceCheckbox.checked,
      centreText: centreTextCheckbox.checked,
      minimumHeight: minimumHeightCheckbox.checked,
      showSubheading: showSubheadingCheckbox.checked,
      h1Serif: h1SerifCheckbox.checked,
      h2Serif: h2SerifCheckbox.checked,
      h1LetterSpacing: Number(h1LetterSpacingSlider.value),
      h2LetterSpacing: Number(h2LetterSpacingSlider.value)
    };

    // Format slider values consistently with their em units.
    function formatLetterSpacing(value) {
      return `${Number(value).toFixed(3)}em`;
    }

    // Update the visible slider values and shared heading CSS rules.
    function updateLetterSpacingStyles() {
      h1SpacingValue.textContent =
        formatLetterSpacing(settings.h1LetterSpacing);

      h2SpacingValue.textContent =
        formatLetterSpacing(settings.h2LetterSpacing);

      heroEffectStylesElement.textContent = `
        .hero h1 {
          letter-spacing: ${settings.h1LetterSpacing}em;
        }

        .hero h2 {
          letter-spacing: ${settings.h2LetterSpacing}em;
        }
      `;
    }

    // Apply current settings without replacing the editable headings.
    function updateHero() {
      settings.darkAppearance = darkAppearanceCheckbox.checked;
      settings.centreText = centreTextCheckbox.checked;
      settings.minimumHeight = minimumHeightCheckbox.checked;
      settings.showSubheading = showSubheadingCheckbox.checked;

      settings.h1Serif = h1SerifCheckbox.checked;
      settings.h2Serif = h2SerifCheckbox.checked;

      settings.h1LetterSpacing = Number(h1LetterSpacingSlider.value);
      settings.h2LetterSpacing = Number(h2LetterSpacingSlider.value);

      heroPreview.classList.toggle(
        "hero-light",
        !settings.darkAppearance
      );

      heroPreview.classList.toggle(
        "hero-dark",
        settings.darkAppearance
      );

      heroPreview.classList.toggle(
        "hero-centre",
        settings.centreText
      );

      heroPreview.classList.toggle(
        "hero-min-height",
        settings.minimumHeight
      );

      heroHeading.classList.toggle(
        "hero-h1-serif",
        settings.h1Serif
      );

      heroSubheading.classList.toggle(
        "hero-h2-serif",
        settings.h2Serif
      );

      // Hiding H2 removes it from layout and keyboard navigation.
      heroSubheading.hidden = !settings.showSubheading;

      // Disable H2 controls without resetting their saved settings.
      h2SerifCheckbox.disabled = !settings.showSubheading;
      h2LetterSpacingSlider.disabled = !settings.showSubheading;

      updateLetterSpacingStyles();
      updateGeneratedHtml();
    }

    // Escape special characters so heading text remains text in HTML.
    function escapeHtml(text) {
      return text
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;");
    }

    // Build the standalone page using the latest settings and heading text.
    function generateHtml() {
      const currentHeading = escapeHtml(heroHeading.textContent);
      const currentSubheading = escapeHtml(heroSubheading.textContent);

      const appearanceClass = settings.darkAppearance
        ? "hero-dark"
        : "hero-light";

      const alignmentClass = settings.centreText
        ? " hero-centre"
        : "";

      const heightClass = settings.minimumHeight
        ? " hero-min-height"
        : "";

      const h1SerifClass = settings.h1Serif
        ? " hero-h1-serif"
        : "";

      const h2SerifClass = settings.h2Serif
        ? " hero-h2-serif"
        : "";

      // Build H2 markup only when the subheading is enabled.
      const subheadingHtml = settings.showSubheading
        ? `
      <h2 class="hero-h2${h2SerifClass}">${currentSubheading}</h2>`
        : "";

      // Escape the closing script tag so it cannot end this script early.
      const tailwindScript =
        '<script src="https://cdn.jsdelivr.net/npm/@tailwindcss/browser@4"><\/script>';

      // Export the shared hero stylesheet, excluding editor-only CSS.
      const exportedHeroStyles = heroStylesElement.textContent;

      // Export the current heading-specific letter-spacing rules.
      const exportedEffectStyles = `
.hero h1 {
  letter-spacing: ${settings.h1LetterSpacing}em;
}

.hero h2 {
  letter-spacing: ${settings.h2LetterSpacing}em;
}`;

      return `<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Hero Section</title>
  <meta name="description" content="Short, unique description of web page. Often used as the snippet shown in search results. About 120 characters is ideal.">
  ${tailwindScript}
  <style type="text/tailwindcss">
${exportedHeroStyles}
  </style>
  <style>
${exportedEffectStyles}
  </style>
</head>
<body>
  <header class="hero ${appearanceClass}${alignmentClass}${heightClass}">
    <div class="hero-content">
      <h1 class="hero-h1${h1SerifClass}">${currentHeading}</h1>${subheadingHtml}
    </div>
  </header>
</body>
</html>`;
    }

    // Keep the manual-copy textarea current after every change.
    function updateGeneratedHtml() {
      generatedHtmlTextarea.value = generateHtml();
    }

    // Display the manual-copy fallback if clipboard access fails.
    function showManualCopyFallback() {
      updateGeneratedHtml();
      fallbackPanel.hidden = false;

      copyStatus.textContent =
        "Automatic copying is unavailable. Select the HTML below and copy it manually.";

      generatedHtmlTextarea.focus();
      generatedHtmlTextarea.select();
    }

    // Copy the latest standalone HTML, handling browser restrictions.
    async function copyHtml() {
      updateGeneratedHtml();

      const htmlToCopy = generatedHtmlTextarea.value;

      if (
        !navigator.clipboard ||
        typeof navigator.clipboard.writeText !== "function"
      ) {
        showManualCopyFallback();
        return;
      }

      try {
        await navigator.clipboard.writeText(htmlToCopy);

        copyStatus.textContent =
          "HTML copied—paste it into VS Code.";

        fallbackPanel.hidden = true;
      } catch (error) {
        showManualCopyFallback();
      }
    }

    // Prevent rich-text pastes from introducing markup into headings.
    function handlePlainTextPaste(event) {
      event.preventDefault();

      const clipboardData = event.clipboardData;

      if (!clipboardData) {
        return;
      }

      const plainText = clipboardData.getData("text/plain");
      const selection = window.getSelection();

      if (!selection || selection.rangeCount === 0) {
        return;
      }

      const activeHeading = document.activeElement;

      if (
        activeHeading !== heroHeading &&
        activeHeading !== heroSubheading
      ) {
        return;
      }

      const range = selection.getRangeAt(0);

      if (!activeHeading.contains(range.commonAncestorContainer)) {
        return;
      }

      range.deleteContents();

      const textNode = document.createTextNode(plainText);
      range.insertNode(textNode);

      // Place the caret immediately after the inserted plain text.
      range.setStartAfter(textNode);
      range.collapse(true);

      selection.removeAllRanges();
      selection.addRange(range);

      updateGeneratedHtml();
    }

    // Listen for changes to layout and typography checkboxes.
    darkAppearanceCheckbox.addEventListener("change", updateHero);
    centreTextCheckbox.addEventListener("change", updateHero);
    minimumHeightCheckbox.addEventListener("change", updateHero);
    showSubheadingCheckbox.addEventListener("change", updateHero);

    h1SerifCheckbox.addEventListener("change", updateHero);
    h2SerifCheckbox.addEventListener("change", updateHero);

    // Update the preview continuously as sliders move.
    h1LetterSpacingSlider.addEventListener("input", updateHero);
    h2LetterSpacingSlider.addEventListener("input", updateHero);

    // Keep exported HTML current when either heading is edited.
    heroHeading.addEventListener("input", updateGeneratedHtml);
    heroSubheading.addEventListener("input", updateGeneratedHtml);

    heroHeading.addEventListener("paste", handlePlainTextPaste);
    heroSubheading.addEventListener("paste", handlePlainTextPaste);

    copyHtmlButton.addEventListener("click", copyHtml);

    // Apply the default settings and generate the initial HTML.
    updateHero();
  </script>
</body>
